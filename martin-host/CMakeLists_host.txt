cmake_minimum_required(VERSION 3.16)
project(mine_host LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT DEFINED ENV{XILINX_XRT})
    message(FATAL_ERROR "XILINX_XRT is not set. Source the XRT setup script first.")
endif()

add_executable(mine_host host.cpp)
target_include_directories(mine_host PRIVATE "$ENV{XILINX_XRT}/include")
target_link_directories(mine_host PRIVATE "$ENV{XILINX_XRT}/lib")
target_link_libraries(mine_host PRIVATE xrt_coreutil pthread)
target_compile_options(mine_host PRIVATE -Wall -Wextra -Wpedantic)
